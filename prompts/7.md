### 1. Create Payment Handler

Create `sdk/src/agentpay/payment.py`:

```python
"""x402 payment handling for AgentPay SDK."""

import hashlib
import json
import time
from typing import Optional
from eth_account import Account
from eth_account.messages import encode_defunct

from .exceptions import ConfigurationError


class X402PaymentHandler:
    """Handle x402 payments on Movement network."""

    def __init__(
        self,
        private_key: str,
        network: str = "movement-testnet",
    ):
        """
        Initialize payment handler.

        Args:
            private_key: Movement wallet private key (hex string with 0x prefix)
            network: Network to use
        """
        if not private_key:
            raise ConfigurationError("Private key is required for payments")

        # Remove 0x prefix if present
        if private_key.startswith("0x"):
            private_key = private_key[2:]

        try:
            self._account = Account.from_key(private_key)
        except Exception as e:
            raise ConfigurationError(f"Invalid private key: {e}")

        self.network = network

    @property
    def address(self) -> str:
        """Get wallet address."""
        return self._account.address

    @property
    def public_key(self) -> str:
        """Get public key."""
        return self._account.key.hex()

    def build_payment_header(
        self,
        pay_to: str,
        amount: str,
        resource: str,
        nonce: Optional[str] = None,
    ) -> str:
        """
        Build x402 payment header.

        Args:
            pay_to: Recipient address
            amount: Amount in octas (smallest unit)
            resource: Facilitator resource URL
            nonce: Optional nonce (generated if not provided)

        Returns:
            Base64-encoded payment header value
        """
        if nonce is None:
            nonce = str(int(time.time() * 1000))

        # Create payment payload
        payload = {
            "x402Version": 1,
            "scheme": "exact",
            "network": self.network.replace("-", "_"),
            "payload": {
                "signature": "",  # Will be filled
                "authorization": {
                    "from": self.address,
                    "to": pay_to,
                    "value": amount,
                    "validAfter": "0",
                    "validBefore": str(int(time.time()) + 3600),  # 1 hour validity
                    "nonce": nonce,
                },
            },
        }

        # Create message to sign
        auth = payload["payload"]["authorization"]
        message = self._create_signing_message(auth)

        # Sign the message
        signed = self._account.sign_message(encode_defunct(text=message))
        payload["payload"]["signature"] = signed.signature.hex()

        # Encode as base64
        import base64
        payload_json = json.dumps(payload, separators=(",", ":"))
        return base64.b64encode(payload_json.encode()).decode()

    def _create_signing_message(self, auth: dict) -> str:
        """Create the message to sign for x402 authorization."""
        # x402 signing message format
        message_parts = [
            f"from:{auth['from']}",
            f"to:{auth['to']}",
            f"value:{auth['value']}",
            f"validAfter:{auth['validAfter']}",
            f"validBefore:{auth['validBefore']}",
            f"nonce:{auth['nonce']}",
        ]
        return "\n".join(message_parts)


class AutoPayClient:
    """Wrapper client that automatically handles x402 payments."""

    def __init__(
        self,
        client: "AgentPayClient",
        payment_handler: X402PaymentHandler,
        max_auto_pay: Optional[str] = None,
    ):
        """
        Initialize auto-pay client wrapper.

        Args:
            client: Base AgentPay client
            payment_handler: Payment handler for signing
            max_auto_pay: Maximum amount to auto-pay (in octas)
        """
        self._client = client
        self._payment = payment_handler
        self._max_auto_pay = int(max_auto_pay) if max_auto_pay else None

    def _should_auto_pay(self, amount: str) -> bool:
        """Check if we should auto-pay this amount."""
        if self._max_auto_pay is None:
            return True
        return int(amount) <= self._max_auto_pay

    def _retry_with_payment(self, method, request, error: "PaymentRequired"):
        """Retry a request with payment header."""
        if not self._should_auto_pay(error.price):
            raise error

        payment_header = self._payment.build_payment_header(
            pay_to=error.pay_to,
            amount=error.price,
            resource=error.resource or "",
        )

        return method(request, payment_header=payment_header)

    def inference(self, request):
        """Run inference with auto-payment."""
        from .exceptions import PaymentRequired
        try:
            return self._client.inference(request)
        except PaymentRequired as e:
            return self._retry_with_payment(self._client.inference, request, e)

    def generate_image(self, request):
        """Generate image with auto-payment."""
        from .exceptions import PaymentRequired
        try:
            return self._client.generate_image(request)
        except PaymentRequired as e:
            return self._retry_with_payment(self._client.generate_image, request, e)

    def run_compute(self, request):
        """Run compute job with auto-payment."""
        from .exceptions import PaymentRequired
        try:
            return self._client.run_compute(request)
        except PaymentRequired as e:
            return self._retry_with_payment(self._client.run_compute, request, e)

    # Delegate non-paid methods
    def health_check(self):
        return self._client.health_check()

    def list_models(self):
        return self._client.list_models()

    def list_providers(self):
        return self._client.list_providers()

    def get_job_status(self, job_id: str):
        return self._client.get_job_status(job_id)

    def cancel_job(self, job_id: str):
        return self._client.cancel_job(job_id)
```

### 2. Update Client for Auto-Pay

Update `sdk/src/agentpay/client.py` constructor:

```python
def __init__(
    self,
    endpoint: Optional[str] = None,
    private_key: Optional[str] = None,
    network: str = "movement-testnet",
    auto_pay: bool = False,
    max_auto_pay: Optional[str] = None,
    timeout: float = 30.0,
):
    # ... existing init code ...

    self._payment_handler = None
    self._auto_pay_wrapper = None

    if private_key and auto_pay:
        from .payment import X402PaymentHandler, AutoPayClient
        self._payment_handler = X402PaymentHandler(private_key, network)
        self._auto_pay_wrapper = AutoPayClient(self, self._payment_handler, max_auto_pay)

def with_auto_pay(self):
    """Get auto-paying version of this client."""
    if self._auto_pay_wrapper is None:
        raise ConfigurationError("Auto-pay not configured. Provide private_key and auto_pay=True")
    return self._auto_pay_wrapper
```

### 4. Create Integration Example

Create `sdk/examples/inference_with_payment.py`:

```python
"""Example: Run inference with automatic x402 payment."""

import os
from agentpay import AgentPayClient, InferenceRequest, ChatMessage

# Initialize client with payment enabled
client = AgentPayClient(
    endpoint=os.environ.get("AGENTPAY_ENDPOINT", "http://localhost:4402"),
    private_key=os.environ["MOVEMENT_PRIVATE_KEY"],
    network="movement-testnet",
    auto_pay=True,
)

# Run inference - payment is automatic!
response = client.with_auto_pay().inference(
    InferenceRequest(
        model="gpt-4o-mini",
        messages=[ChatMessage(role="user", content="Hello, how are you?")],
        max_tokens=256,
    )
)

print(f"Response: {response.choices[0].message.content}")
print(f"Tokens: {response.usage.total_tokens}")
if response.cost:
    print(f"Cost: {response.cost.amount} {response.cost.currency}")
```

### 5. Create Manual Payment Example

Create `sdk/examples/manual_payment.py`:

```python
"""Example: Handle payments manually."""

import os
from agentpay import (
    AgentPayClient,
    InferenceRequest,
    ChatMessage,
    X402PaymentHandler,
    PaymentRequired,
)

# Client without auto-pay
client = AgentPayClient(
    endpoint=os.environ.get("AGENTPAY_ENDPOINT", "http://localhost:4402"),
)

# Separate payment handler
payment = X402PaymentHandler(
    private_key=os.environ["MOVEMENT_PRIVATE_KEY"],
    network="movement-testnet",
)

print(f"Wallet address: {payment.address}")

request = InferenceRequest(
    model="gpt-4o-mini",
    messages=[ChatMessage(role="user", content="What is 2+2?")],
)

try:
    # First attempt - will fail with 402
    response = client.inference(request)
except PaymentRequired as e:
    print(f"Payment required: {e.price} {e.currency}")
    print(f"Pay to: {e.pay_to}")

    # Build payment header
    header = payment.build_payment_header(
        pay_to=e.pay_to,
        amount=e.price,
        resource=e.resource or "",
    )

    # Retry with payment
    response = client.inference(request, payment_header=header)

print(f"Response: {response.choices[0].message.content}")
```
