"""Data models for AgentPay SDK."""

from typing import Optional, List
from pydantic import BaseModel, Field


class ChatMessage(BaseModel):
    """A chat message."""
    role: str = Field(..., description="Message role: system, user, or assistant")
    content: str = Field(..., description="Message content")


class InferenceRequest(BaseModel):
    """Request for LLM inference."""
    model: str = Field(..., description="Model ID (e.g., 'gpt-4o-mini', 'llama-3.1-70b')")
    messages: Optional[List[ChatMessage]] = None
    prompt: Optional[str] = None
    max_tokens: int = Field(default=512, ge=1, le=4096)
    temperature: float = Field(default=0.7, ge=0, le=2)
    stream: bool = False
    provider: Optional[str] = None
    max_latency_ms: Optional[int] = None
    max_price: Optional[float] = None


class Usage(BaseModel):
    """Token usage statistics."""
    prompt_tokens: int
    completion_tokens: int
    total_tokens: int


class Cost(BaseModel):
    """Cost information."""
    amount: str
    currency: str = "octas"
    total_cost: Optional[str] = None
    tx_hash: Optional[str] = None


class Choice(BaseModel):
    """Inference response choice."""
    message: ChatMessage
    finish_reason: Optional[str] = None


class InferenceResponse(BaseModel):
    """Response from LLM inference."""
    id: str
    model: str
    provider: str
    choices: List[Choice]
    usage: Usage
    cost: Optional[Cost] = None
    latency_ms: Optional[int] = None


class ImageGenerationRequest(BaseModel):
    """Request for image generation."""
    prompt: str
    model: str = "sdxl"
    size: str = "1024x1024"
    n: int = 1


class ImageData(BaseModel):
    """Generated image data."""
    url: Optional[str] = None
    b64_json: Optional[str] = None


class ImageGenerationResponse(BaseModel):
    """Response from image generation."""
    data: List[ImageData]
    cost: Optional[Cost] = None


class GpuConfig(BaseModel):
    """GPU configuration for compute jobs."""
    type: str = "A100"
    count: int = 1


class ComputeRunRequest(BaseModel):
    """Request to run a compute job."""
    image: str
    command: List[str]
    env: Optional[dict] = None
    gpu: Optional[GpuConfig] = None
    timeout_seconds: int = 3600


class ComputeJob(BaseModel):
    """Compute job status."""
    id: str
    status: str  # pending, running, completed, failed
    output: Optional[str] = None
    logs: Optional[str] = None
    cost: Optional[Cost] = None


class ModelInfo(BaseModel):
    """Information about an available model."""
    id: str
    provider: str
    price_per_unit: str
    unit: str = "1K tokens"


class ProviderInfo(BaseModel):
    """Information about a provider."""
    id: str
    name: str
    status: str
    latency_ms: Optional[int] = None
    availability: Optional[float] = None
    models: List[str] = []
```

### 3. Create Exceptions

Create `sdk/src/agentpay/exceptions.py`:

"""Exceptions for AgentPay SDK."""

from typing import Optional


class AgentPayError(Exception):
    """Base exception for AgentPay SDK."""
    pass


class APIError(AgentPayError):
    """API returned an error."""
    def __init__(self, message: str, code: Optional[int] = None):
        self.message = message
        self.code = code
        super().__init__(f"[{code}] {message}" if code else message)


class PaymentRequired(AgentPayError):
    """Payment is required (402 response)."""
    def __init__(
        self,
        price: str,
        currency: str,
        network: str,
        pay_to: str,
        resource: Optional[str] = None,
    ):
        self.price = price
        self.currency = currency
        self.network = network
        self.pay_to = pay_to
        self.resource = resource
        super().__init__(f"Payment required: {price} {currency} on {network}")


class JobNotFoundError(AgentPayError):
    """Compute job not found."""
    def __init__(self, job_id: str):
        self.job_id = job_id
        super().__init__(f"Job not found: {job_id}")


class ConfigurationError(AgentPayError):
    """Configuration error."""
    pass
```

### 4. Create Core Client

Create `sdk/src/agentpay/client.py`:

```python
"""AgentPay client for making API requests."""

import os
from typing import Optional, Iterator, List
import httpx

from .models import (
    InferenceRequest,
    InferenceResponse,
    ImageGenerationRequest,
    ImageGenerationResponse,
    ComputeRunRequest,
    ComputeJob,
    ModelInfo,
    ProviderInfo,
    ChatMessage,
    Choice,
    Usage,
    Cost,
)
from .exceptions import APIError, PaymentRequired, JobNotFoundError


class AgentPayClient:
    """Client for interacting with AgentPay API."""

    def __init__(
        self,
        endpoint: Optional[str] = None,
        private_key: Optional[str] = None,
        network: str = "movement-testnet",
        auto_pay: bool = False,
        timeout: float = 30.0,
    ):
        """
        Initialize AgentPay client.

        Args:
            endpoint: API endpoint URL
            private_key: Movement wallet private key for payments
            network: Network to use (movement-testnet or movement-mainnet)
            auto_pay: Whether to automatically pay on 402 responses
            timeout: Request timeout in seconds
        """
        self.endpoint = endpoint or os.environ.get(
            "AGENTPAY_ENDPOINT", "http://localhost:4402"
        )
        self.private_key = private_key or os.environ.get("MOVEMENT_PRIVATE_KEY")
        self.network = network
        self.auto_pay = auto_pay
        self.timeout = timeout

        self._client = httpx.Client(
            base_url=self.endpoint,
            timeout=timeout,
            headers={"Content-Type": "application/json"},
        )

    def close(self):
        """Close the HTTP client."""
        self._client.close()

    def __enter__(self):
        return self

    def __exit__(self, *args):
        self.close()

    def _handle_response(self, response: httpx.Response) -> dict:
        """Handle API response, raising appropriate exceptions."""
        if response.status_code == 402:
            data = response.json()
            raise PaymentRequired(
                price=data.get("price", "0"),
                currency=data.get("currency", "octas"),
                network=data.get("network", "movement"),
                pay_to=data.get("payTo", ""),
                resource=data.get("resource"),
            )

        if response.status_code == 404:
            data = response.json()
            if "job" in data.get("error", "").lower():
                raise JobNotFoundError(data.get("jobId", "unknown"))

        if not response.is_success:
            try:
                data = response.json()
                raise APIError(data.get("error", "Unknown error"), response.status_code)
            except ValueError:
                raise APIError(response.text, response.status_code)

        return response.json()

    def health_check(self) -> bool:
        """Check if the server is healthy."""
        try:
            response = self._client.get("/health")
            return response.is_success
        except Exception:
            return False

    def list_models(self) -> List[ModelInfo]:
        """List available models."""
        response = self._client.get("/v1/models")
        data = self._handle_response(response)
        return [ModelInfo(**m) for m in data.get("models", [])]

    def list_providers(self) -> List[ProviderInfo]:
        """List available providers."""
        response = self._client.get("/v1/providers")
        data = self._handle_response(response)
        return [ProviderInfo(**p) for p in data.get("providers", [])]

    def inference(
        self,
        request: InferenceRequest,
        payment_header: Optional[str] = None,
    ) -> InferenceResponse:
        """
        Run LLM inference.

        Args:
            request: Inference request parameters
            payment_header: X-PAYMENT header value (for manual payment)

        Returns:
            Inference response with generated text

        Raises:
            PaymentRequired: If payment is needed and auto_pay is False
            APIError: If API returns an error
        """
        headers = {}
        if payment_header:
            headers["X-PAYMENT"] = payment_header

        response = self._client.post(
            "/v1/inference",
            json=request.model_dump(exclude_none=True),
            headers=headers,
        )
        data = self._handle_response(response)

        # Convert response to model
        return InferenceResponse(
            id=data.get("id", ""),
            model=data.get("model", request.model),
            provider=data.get("provider", ""),
            choices=[
                Choice(
                    message=ChatMessage(
                        role="assistant",
                        content=data.get("output", ""),
                    ),
                    finish_reason="stop",
                )
            ],
            usage=Usage(**data.get("usage", {"prompt_tokens": 0, "completion_tokens": 0, "total_tokens": 0})),
            cost=Cost(**data["cost"]) if data.get("cost") else None,
            latency_ms=data.get("latency_ms"),
        )

    def chat(
        self,
        model: str,
        messages: List[ChatMessage],
        max_tokens: int = 512,
        temperature: float = 0.7,
        payment_header: Optional[str] = None,
    ) -> InferenceResponse:
        """Convenience method for chat-style inference."""
        request = InferenceRequest(
            model=model,
            messages=messages,
            max_tokens=max_tokens,
            temperature=temperature,
        )
        return self.inference(request, payment_header)

    def generate_image(
        self,
        request: ImageGenerationRequest,
        payment_header: Optional[str] = None,
    ) -> ImageGenerationResponse:
        """Generate an image."""
        headers = {}
        if payment_header:
            headers["X-PAYMENT"] = payment_header

        response = self._client.post(
            "/v1/images/generate",
            json=request.model_dump(exclude_none=True),
            headers=headers,
        )
        data = self._handle_response(response)
        return ImageGenerationResponse(**data)

    def run_compute(
        self,
        request: ComputeRunRequest,
        payment_header: Optional[str] = None,
    ) -> ComputeJob:
        """Start a compute job."""
        headers = {}
        if payment_header:
            headers["X-PAYMENT"] = payment_header

        response = self._client.post(
            "/v1/compute/run",
            json=request.model_dump(exclude_none=True),
            headers=headers,
        )
        data = self._handle_response(response)
        return ComputeJob(**data)

    def get_job_status(self, job_id: str) -> ComputeJob:
        """Get status of a compute job."""
        response = self._client.get(f"/v1/compute/status/{job_id}")
        data = self._handle_response(response)
        return ComputeJob(**data)

    def cancel_job(self, job_id: str) -> bool:
        """Cancel a compute job."""
        response = self._client.post(f"/v1/compute/cancel/{job_id}")
        if response.status_code == 404:
            raise JobNotFoundError(job_id)
        return response.is_success
```
