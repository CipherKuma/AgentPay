"""Tests for AgentPay client."""

import pytest
from pytest_httpx import HTTPXMock

from agentpay import (
    AgentPayClient,
    InferenceRequest,
    ChatMessage,
    PaymentRequired,
    APIError,
    JobNotFoundError,
)


class TestClientInit:
    def test_default_endpoint(self):
        client = AgentPayClient()
        assert "localhost:4402" in client.endpoint

    def test_custom_endpoint(self):
        client = AgentPayClient(endpoint="http://custom:8080")
        assert client.endpoint == "http://custom:8080"


class TestHealthCheck:
    def test_healthy_server(self, client, httpx_mock: HTTPXMock):
        httpx_mock.add_response(url="http://test-server:4402/health", status_code=200)
        assert client.health_check() is True

    def test_unhealthy_server(self, client, httpx_mock: HTTPXMock):
        httpx_mock.add_response(url="http://test-server:4402/health", status_code=503)
        assert client.health_check() is False

    def test_connection_error(self, client, httpx_mock: HTTPXMock):
        httpx_mock.add_exception(Exception("Connection refused"))
        assert client.health_check() is False


class TestListModels:
    def test_success(self, client, httpx_mock: HTTPXMock, mock_models_response):
        httpx_mock.add_response(
            url="http://test-server:4402/v1/models",
            json=mock_models_response,
        )
        models = client.list_models()
        assert len(models) == 2
        assert models[0].id == "gpt-4o-mini"
        assert models[1].provider == "together"


class TestInference:
    def test_success(self, client, httpx_mock: HTTPXMock, mock_inference_response):
        httpx_mock.add_response(
            url="http://test-server:4402/v1/inference",
            json=mock_inference_response,
        )
        response = client.inference(
            InferenceRequest(
                model="gpt-4o-mini",
                messages=[ChatMessage(role="user", content="Hello")],
            )
        )
        assert response.id == "test-123"
        assert response.provider == "openai"
        assert response.usage.total_tokens == 25

    def test_payment_required(self, client, httpx_mock: HTTPXMock, mock_402_response):
        httpx_mock.add_response(
            url="http://test-server:4402/v1/inference",
            status_code=402,
            json=mock_402_response,
        )
        with pytest.raises(PaymentRequired) as exc_info:
            client.inference(InferenceRequest(model="test", prompt="test"))

        assert exc_info.value.price == "5000"
        assert exc_info.value.currency == "octas"
        assert exc_info.value.network == "movement-testnet"

    def test_with_payment_header(self, client, httpx_mock: HTTPXMock, mock_inference_response):
        httpx_mock.add_response(
            url="http://test-server:4402/v1/inference",
            json=mock_inference_response,
        )
        response = client.inference(
            InferenceRequest(model="test", prompt="test"),
            payment_header="base64-payment-data",
        )

        # Verify header was sent
        request = httpx_mock.get_requests()[0]
        assert request.headers.get("X-PAYMENT") == "base64-payment-data"

    def test_api_error(self, client, httpx_mock: HTTPXMock):
        httpx_mock.add_response(
            url="http://test-server:4402/v1/inference",
            status_code=500,
            json={"error": "Internal server error"},
        )
        with pytest.raises(APIError) as exc_info:
            client.inference(InferenceRequest(model="test", prompt="test"))

        assert exc_info.value.code == 500


class TestChat:
    def test_convenience_method(self, client, httpx_mock: HTTPXMock, mock_inference_response):
        httpx_mock.add_response(
            url="http://test-server:4402/v1/inference",
            json=mock_inference_response,
        )
        response = client.chat(
            model="gpt-4o-mini",
            messages=[ChatMessage(role="user", content="Hi")],
        )
        assert response.choices[0].message.content is not None


class TestComputeJobs:
    def test_job_not_found(self, client, httpx_mock: HTTPXMock):
        httpx_mock.add_response(
            url="http://test-server:4402/v1/compute/status/invalid-id",
            status_code=404,
            json={"error": "Job not found", "jobId": "invalid-id"},

### 4. Create Payment Tests

Create `sdk/tests/test_payment.py`:

```python
"""Tests for payment handling."""

import pytest
import json
import base64

from agentpay import X402PaymentHandler, ConfigurationError


# Test private key (DO NOT USE IN PRODUCTION)
TEST_PRIVATE_KEY = "0x" + "a" * 64  # 32 bytes of 0xaa


class TestX402PaymentHandler:
    def test_init_with_valid_key(self):
        handler = X402PaymentHandler(TEST_PRIVATE_KEY)
        assert handler.address.startswith("0x")
        assert len(handler.address) == 42

    def test_init_without_0x_prefix(self):
        handler = X402PaymentHandler("a" * 64)
        assert handler.address.startswith("0x")

    def test_init_with_invalid_key(self):
        with pytest.raises(ConfigurationError):
            X402PaymentHandler("invalid-key")

    def test_init_with_empty_key(self):
        with pytest.raises(ConfigurationError):
            X402PaymentHandler("")

    def test_build_payment_header(self):
        handler = X402PaymentHandler(TEST_PRIVATE_KEY)
        header = handler.build_payment_header(
            pay_to="0x1234567890abcdef1234567890abcdef12345678",
            amount="5000",
            resource="https://facilitator.x402.org",
        )

        # Verify it's valid base64
        decoded = base64.b64decode(header)
        payload = json.loads(decoded)

        assert payload["x402Version"] == 1
        assert payload["scheme"] == "exact"
        assert "signature" in payload["payload"]
        assert payload["payload"]["authorization"]["value"] == "5000"

    def test_build_payment_header_with_nonce(self):
        handler = X402PaymentHandler(TEST_PRIVATE_KEY)
        header = handler.build_payment_header(
            pay_to="0x1234567890abcdef1234567890abcdef12345678",
            amount="5000",
            resource="https://facilitator.x402.org",
            nonce="12345",
        )

        decoded = base64.b64decode(header)
        payload = json.loads(decoded)

        assert payload["payload"]["authorization"]["nonce"] == "12345"


class TestAutoPayClient:
    @pytest.fixture
    def client_with_payment(self):
        from agentpay import AgentPayClient
        return AgentPayClient(
            endpoint="http://test:4402",
            private_key=TEST_PRIVATE_KEY,
            auto_pay=True,
        )

    def test_with_auto_pay_returns_wrapper(self, client_with_payment):
        wrapper = client_with_payment.with_auto_pay()
        assert wrapper is not None

    def test_with_auto_pay_without_config_raises(self):
        from agentpay import AgentPayClient
        client = AgentPayClient(endpoint="http://test:4402")
        with pytest.raises(ConfigurationError):
            client.with_auto_pay()
```

### 5. Create Integration Tests

Create `sdk/tests/test_integration.py`:

```python
"""Integration tests - require running server."""

import os
import pytest

from agentpay import AgentPayClient, InferenceRequest, ChatMessage, PaymentRequired


# Skip if no server endpoint configured
pytestmark = pytest.mark.skipif(
    not os.environ.get("AGENTPAY_ENDPOINT"),
    reason="AGENTPAY_ENDPOINT not set"
)


@pytest.fixture
def live_client():
    return AgentPayClient(
        endpoint=os.environ.get("AGENTPAY_ENDPOINT"),
        timeout=30.0,
    )


class TestLiveServer:
    def test_health_check(self, live_client):
        assert live_client.health_check() is True

    def test_list_models(self, live_client):
        models = live_client.list_models()
        assert len(models) > 0
        assert all(m.id for m in models)

    def test_list_providers(self, live_client):
        providers = live_client.list_providers()
        assert len(providers) > 0

    def test_inference_requires_payment(self, live_client):
        """Without payment, should get 402."""
        with pytest.raises(PaymentRequired):
            live_client.inference(
                InferenceRequest(
                    model="gpt-4o-mini",
                    prompt="Hello",
                )
            )


@pytest.mark.skipif(
    not os.environ.get("MOVEMENT_PRIVATE_KEY"),
    reason="MOVEMENT_PRIVATE_KEY not set"
)
class TestLivePayment:
    @pytest.fixture
    def paying_client(self):
        return AgentPayClient(
            endpoint=os.environ.get("AGENTPAY_ENDPOINT"),
            private_key=os.environ.get("MOVEMENT_PRIVATE_KEY"),
            auto_pay=True,
            timeout=60.0,
        )

    def test_inference_with_payment(self, paying_client):
        """Full inference with automatic payment."""
        response = paying_client.with_auto_pay().inference(
            InferenceRequest(
                model="gpt-4o-mini",
                messages=[ChatMessage(role="user", content="Say hello")],
                max_tokens=50,
            )
        )
        assert response.choices[0].message.content
        assert response.usage.total_tokens > 0
```
