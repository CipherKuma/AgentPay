### 2. Run Integration Tests

Create `tests/run-e2e.sh`:

```bash
#!/bin/bash
set -e

echo "=== AgentPay E2E Tests ==="

# Colors
GREEN='\033[0;32m'
RED='\033[0;31m'
NC='\033[0m'

# Configuration
SERVER_URL=${AGENTPAY_ENDPOINT:-http://localhost:4402}
DASHBOARD_URL=${DASHBOARD_URL:-http://localhost:3000}

pass() { echo -e "${GREEN}✓ $1${NC}"; }
fail() { echo -e "${RED}✗ $1${NC}"; exit 1; }

# Test health endpoint
echo "Testing health endpoint..."
curl -sf "$SERVER_URL/health" > /dev/null && pass "Health check" || fail "Health check failed"

# Test models endpoint
echo "Testing models endpoint..."
MODELS=$(curl -sf "$SERVER_URL/v1/models")
echo "$MODELS" | jq -e '.models | length > 0' > /dev/null && pass "Models endpoint" || fail "No models returned"

# Test providers endpoint
echo "Testing providers endpoint..."
PROVIDERS=$(curl -sf "$SERVER_URL/v1/providers")
echo "$PROVIDERS" | jq -e '.providers | length > 0' > /dev/null && pass "Providers endpoint" || fail "No providers returned"

# Test inference requires payment
echo "Testing inference payment requirement..."
RESPONSE=$(curl -s -w "\n%{http_code}" -X POST "$SERVER_URL/v1/inference" \
  -H "Content-Type: application/json" \
  -d '{"model": "gpt-4o-mini", "prompt": "test"}')
HTTP_CODE=$(echo "$RESPONSE" | tail -1)
[ "$HTTP_CODE" = "402" ] && pass "Inference requires payment" || fail "Expected 402, got $HTTP_CODE"

# Test 402 response format
BODY=$(echo "$RESPONSE" | head -n -1)
echo "$BODY" | jq -e '.price and .currency and .network and .payTo' > /dev/null \
  && pass "402 response format" || fail "Invalid 402 response"

# Test usage endpoint
echo "Testing usage endpoint..."
curl -sf "$SERVER_URL/v1/usage" > /dev/null && pass "Usage endpoint" || fail "Usage endpoint failed"

# Test OpenAPI docs
echo "Testing OpenAPI docs..."
curl -sf "$SERVER_URL/docs" > /dev/null && pass "Swagger UI" || fail "Swagger UI not available"
curl -sf "$SERVER_URL/openapi.yaml" > /dev/null && pass "OpenAPI spec" || fail "OpenAPI spec not available"

echo ""
echo "=== All tests passed! ==="
```

#### Environment Validation

Add startup validation in `server/src/index.ts`:

```typescript
function validateEnvironment() {
  const warnings: string[] = [];
  const errors: string[] = [];

  // Required for any functionality
  if (!config.movement.rpcUrl) {
    warnings.push('MOVEMENT_RPC_URL not set, using default');
  }

  // Required for providers
  if (!config.openai.apiKey && !config.together.apiKey && !config.groq.apiKey) {
    errors.push('No provider API keys configured. Set at least one of: OPENAI_API_KEY, TOGETHER_API_KEY, GROQ_API_KEY');
  }

  // Optional but recommended
  if (!config.database.url) {
    warnings.push('DATABASE_URL not set, using in-memory storage');
  }

  warnings.forEach(w => console.warn(`[Config Warning] ${w}`));
  if (errors.length > 0) {
    errors.forEach(e => console.error(`[Config Error] ${e}`));
    process.exit(1);
  }
}

validateEnvironment();
```
